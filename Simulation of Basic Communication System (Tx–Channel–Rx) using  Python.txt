from scipy.io import wavfile

fs, audio = wavfile.read("/content/drive/MyDrive/mdm exp/hello.wav")

audio = audio.astype(float)
# Normalize
maxval = max(abs(audio))
audio = audio / maxval

from IPython.display import Audio
Audio("/content/drive/MyDrive/mdm exp/hello.wav")

import numpy as np

t = np.arange(len(audio)) / fs

carrier_frequency = 20000

carrier = np.cos(2*np.pi*carrier_frequency*t)

modulated = (1 + audio) * carrier

# High-frequency interference
noise_freq = 12000      # Hz
noise_amp = 0.15       # Amplitude

hf_noise = noise_amp * np.sin(2 * np.pi * noise_freq * t)

received = modulated + hf_noise

envelope = np.abs(received)
window = 200

demodulated = np.convolve(
    envelope,
    np.ones(window)/window,
    mode='same'
)

wavfile.write(
    "recovered.wav",
    fs,
  8*(demodulated*maxval).astype(np.int16)
)
from IPython.display import Audio
Audio("recovered.wav")

import matplotlib.pyplot as plt

plt.figure(figsize=(12,10))

plt.subplot(6,1,1)
plt.plot(audio)
plt.title("Original Voice Signal")

plt.subplot(6,1,2)
plt.plot(carrier)
plt.title("carrier Signal")


plt.subplot(6,1,3)
plt.plot(modulated)
plt.title("AM Modulated Signal")

plt.subplot(6,1,4)
plt.plot(received)
plt.title("Received Signal with Noise")

plt.subplot(6,1,5)
plt.plot(envelope)
plt.title("Envelope Detector Output")

plt.subplot(6,1,6)
plt.plot(demodulated)
plt.title("Recovered Voice Signal")

plt.tight_layout()
plt.show()
